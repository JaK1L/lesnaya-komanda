```bash
# Вариант 2: Напрямую через psql
psql $DATABASE_URL < backend/migrations/add_performance_indexes.sql

# Вариант 3: Через Python
python -c "
import asyncio
import asyncpg
import os
from dotenv import load_dotenv

load_dotenv()

async def apply():
    conn = await asyncpg.connect(os.getenv('DATABASE_URL'))
    with open('backend/migrations/add_performance_indexes.sql', 'r') as f:
        await conn.execute(f.read())
    await conn.execute('ANALYZE')
    await conn.close()
    print('✅ Индексы применены!')

asyncio.run(apply())
"
```
